function initData() { const hash = window.location.hash; if (hash.startsWith('#data=')) { try { const rawData = hash.replace('#data=', ''); if (!rawData) throw new Error("Hash 資料為空"); const jsonStr = safeAtob(rawData); const sharedTrip = JSON.parse(jsonStr); let localTrips = JSON.parse(localStorage.getItem('travel_split_trips')) || defaultTrips; const index = localTrips.findIndex(t => t.id === sharedTrip.id); if (index !== -1) { localTrips[index] = sharedTrip; } else { localTrips.unshift(sharedTrip); } trips = localTrips; currentTripId = sharedTrip.id; saveToLocalStorage(); } catch (e) { console.error('網址資料解析失敗，已自動重置 Hash：', e); // 當解析失敗時，自動清空錯誤的 Hash，避免整頁卡死 window.history.replaceState(null, '', window.location.pathname); loadFromLocal(); } } else { loadFromLocal(); } }
